# React Integration Guide

<FrameworkBadge framework="react" status="active" />

> Copy the code on this page into your own project. There is no React package to
> install: the SDK is injected by the platform, and the provider below is yours
> to own and adapt.

React is the best-supported framework for Sewa mini apps. The pattern below covers the whole lifecycle: reading the injected SDK from `window.__SEWA_SDK__`, exposing it through a provider, and cleaning up on unmount.

## 2. Entry Points

### `main.tsx` - host-callable mount

```typescript
// src/main.tsx
import { createRoot } from "react-dom/client";
import type { Root } from "react-dom/client";
import App from "./App.tsx";

export function mount(container: HTMLElement, runtime?: { initialPath?: string }) {
  const root: Root = createRoot(container);
  root.render(<App initialPath={runtime?.initialPath} />);
  return { unmount() { root.unmount(); } };
}
```

### `App.tsx` - MemoryRouter + SDK provider

```typescript
// src/App.tsx
import { MemoryRouter } from "react-router";
import { ErrorBoundary } from "./components/ErrorBoundary";
import { PlatformSDKProvider } from "./providers/PlatformSDKProvider";
import TestMiniApp from "./components/MiniApp";

function resolveInitialEntry(initialPath?: string) {
  const hash = window.location.hash.replace(/^#/, "");
  return hash || initialPath || "/";
}

export default function App({ initialPath }: { initialPath?: string }) {
  return (
    <ErrorBoundary>
      <MemoryRouter initialEntries={[resolveInitialEntry(initialPath)]}>
        <PlatformSDKProvider>
          <TestMiniApp />
        </PlatformSDKProvider>
      </MemoryRouter>
    </ErrorBoundary>
  );
}
```

Use `MemoryRouter` (not `BrowserRouter`) - the mini app runs inside a host container and must not touch the browser history.

## 3. SDK Context & Provider

The provider reads the host-injected SDK from `window.__SEWA_SDK__`, retries while the SDK is still connecting, and aborts cleanly on unmount. Its state machine is `loading → ready | error`, so your UI always knows which of the three it is in.

Key pieces:

```typescript
// src/context/SDKContext.tsx
export interface SDKContextValue {
  sdk: MiniAppSdk | null;
  user: SdkPlatformUser | null;
  isReady: boolean;
  error: Error | null;
}
export const SDKContext = createContext<SDKContextValue>({
  sdk: null, user: null, isReady: false, error: null,
});

// src/providers/PlatformSDKProvider.tsx (simplified)
function getSDK(): MiniAppSdk {
  if (typeof window !== "undefined" && window.__SEWA_SDK__) return window.__SEWA_SDK__;
  return createStandaloneMockSDK(); // localStorage-backed mock for dev
}

export function PlatformSDKProvider({ children }: { children: ReactNode }) {
  const [state, dispatch] = useReducer(reducer, { phase: "loading" });
  const abortRef = useRef<AbortController | null>(null);

  useEffect(() => {
    const controller = new AbortController();
    abortRef.current = controller;
    const init = async () => {
      try {
        const instance = await retry(() => getSDK(), { maxAttempts: 10, delayMs: 200, signal: controller.signal });
        const user = await instance.auth.getUser();
        if (!controller.signal.aborted) dispatch({ type: "ready", sdk: instance, user: user ?? null });
      } catch (error) {
        if (!controller.signal.aborted) dispatch({ type: "error", error: error as Error });
      }
    };
    init();
    return () => controller.abort();
  }, []);

  if (state.phase === "error") return <LoadError message={state.error.message} />;
  if (state.phase === "loading") return <div>Connecting to platform...</div>;

  return (
    <SDKContext.Provider value={{ sdk: state.sdk, user: state.user, isReady: true, error: null }}>
      <PlatformSDKRuntime>{children}</PlatformSDKRuntime>
    </SDKContext.Provider>
  );
}
```

`PlatformSDKRuntime` calls `useHostRouter()` inside the provider so back-button sync is always active.

## 4. Hooks

### `usePlatformSDK` - guarded access

```typescript
// src/hooks/usePlatformSDK.ts
export function usePlatformSDK(): SDKContextValue & { sdk: MiniAppSdk } {
  const ctx = useContext(SDKContext);
  if (!ctx.sdk) throw new Error("usePlatformSDK must be used within PlatformSDKProvider");
  return { ...ctx, sdk: ctx.sdk };
}
```

### `useHostRouter` - bidirectional navigation sync

Keep the host's back-button policy in sync and handle `navigation.back.requested`:

```typescript
// src/hooks/useHostRouter.ts
export const NAVIGATION_EVENTS = {
  BACK_REQUESTED: "navigation.back.requested",
  ROUTE_CHANGED: "navigation.route.changed",
} as const;

export function useHostRouter() {
  const { sdk, isReady } = usePlatformSDK();
  const navigate = useNavigate();
  const location = useLocation();
  const navigationType = useNavigationType();
  const depthRef = useRef(0);

  // Track history depth via PUSH/POP + emit ROUTE_CHANGED
  useEffect(() => {
    if (navigationType === "PUSH") depthRef.current += 1;
    else if (navigationType === "POP") depthRef.current = Math.max(0, depthRef.current - 1);
    sdk?.emit(NAVIGATION_EVENTS.ROUTE_CHANGED, {
      previous: previousPathRef.current,
      current: location.pathname,
      canGoBack: depthRef.current > 0,
    });
    if (navigationType === "PUSH") void sdk?.navigation.router.push(true);
  }, [location.key, navigationType, sdk]);

  const goBack = useCallback(async () => {
    const consumed = depthRef.current > 0;
    if (consumed) navigate(-1);
    await sdk?.navigation.router.back(consumed);
  }, [navigate, sdk]);

  useEffect(() => {
    if (!sdk || !isReady) return;
    return sdk.on(NAVIGATION_EVENTS.BACK_REQUESTED, () => { void goBack(); });
  }, [sdk, isReady, goBack]);
}
```

## 5. Vite Build

```typescript
// vite.config.ts - ES library consumed by the host
export default defineConfig({
  plugins: [react(), tailwindcss(), createManifest()],
  build: {
    target: "chrome89",
    outDir: "dist",
    lib: {
      entry: "./src/main.tsx",
      name: "RevenueMiniApp",
      formats: ["es"],
      fileName: () => "[name][hash].js",
    },
    rollupOptions: { output: { format: "es", chunkFileNames: "assets/[name]-[hash].js" } },
  },
});
```

The `createManifest` plugin emits the discovery manifest the platform reads when loading your bundle.
