# React Integration Guide

> **Package:** `@sewa/framework-react` - internal workspace package at `packages/framework-react`. This guide is extracted from the reference app at [`test-mini-app/`](/).

React is the production-ready framework for Sewa mini apps. The reference implementation in `test-mini-app` shows the exact pattern: a Vite ES-library build, host-injected SDK via `window.__GSA_SDK__`, and a `PlatformSDKProvider` with strict lifecycle handling.

## 2. Entry Points

### `main.tsx` - host-callable mount

```typescript
// src/main.tsx
import type { Root } from "react-dom/client";
import { createRoot } from "react-dom/client";
import App from "./App.tsx";
import "./index.css";

/** Sets the initial hash so HashRouter starts at the host-provided path. */
function applyInitialHash(initialPath?: string) {
  if (!initialPath) return;
  const normalized = initialPath.startsWith("/") ? initialPath : `/${initialPath}`;
  if (
    !window.location.hash ||
    window.location.hash === "#" ||
    !window.location.hash.startsWith("#/")
  ) {
    window.location.replace(`#${normalized}`);
  }
}

export function mount(
  container: HTMLElement,
  runtime?: { initialPath?: string },
) {
  const root: Root = createRoot(container);
  applyInitialHash(runtime?.initialPath);
  root.render(<App />);
  return { unmount() { root.unmount(); } };
}
```

### `App.tsx` - HashRouter + SDK provider

```typescript
// src/App.tsx
import { HashRouter } from "react-router";
import { ErrorBoundary } from "./components/ErrorBoundary";
import { PlatformSDKProvider } from "./providers/PlatformSDKProvider";
import TestMiniApp from "./components/MiniApp";
import "./index.css";

export default function App() {
  return (
    <ErrorBoundary>
      <HashRouter>
        <PlatformSDKProvider>
          <TestMiniApp />
        </PlatformSDKProvider>
      </HashRouter>
    </ErrorBoundary>
  );
}
```

The mini app uses `HashRouter` - the host sets `window.location.hash` to the initial path (`main.tsx`'s `applyInitialHash` normalizes `runtime.initialPath` into the hash). This avoids touching browser history in a way that would break the host container's back-button policy. If your app runs inside a Flutter/native shell, `HashRouter` keeps navigation self-contained.

## 3. SDK Context & Provider

The provider in `test-mini-app/src/providers/PlatformSDKProvider.tsx` reads the host-injected SDK from `window.__GSA_SDK__` with retry, an `AbortController` for clean unmount, and a reducer-based state machine (`loading → ready | error`). A full standalone mock is included so the app also runs outside the host during development.

Key pieces:

```typescript
// src/context/SDKContext.tsx
export interface SDKContextValue {
  sdk: MiniAppSdk | null;
  user: SdkPlatformUser | null;
  isReady: boolean;
  error: Error | null;
}
export const SDKContext = createContext<SDKContextValue>({
  sdk: null, user: null, isReady: false, error: null,
});

 // src/providers/PlatformSDKProvider.tsx
function getSDK(): MiniAppSdk {
  const instance = window.__GSA_SDK__ ?? null;
  if (!instance) throw new Error("SDK not available");
  return instance;
}

export function PlatformSDKProvider({ children }: { children: ReactNode }) {
  const [state, dispatch] = useReducer(reducer, { phase: "loading" });
  const abortRef = useRef<AbortController | null>(null);

  useEffect(() => {
    const controller = new AbortController();
    abortRef.current = controller;
    const init = async () => {
      try {
        const instance = await retry(() => getSDK(), {
          maxAttempts: 30, delayMs: 500, signal: controller.signal,
        });
        // Wait for the host to signal that the SDK bridge finished wiring.
        await new Promise<void>((res) => {
          if (window.__GSA_SDK__) return res();
          window.addEventListener("__GSA_SDK_READY__", () => res(), { once: true });
        });
        const user = await instance.auth.getUser();
        if (!controller.signal.aborted) {
          dispatch({ type: "ready", sdk: instance, user: user ?? null });
        }
      } catch (error) {
        if (!controller.signal.aborted) {
          dispatch({
            type: "error",
            error: error instanceof Error ? error : new Error("SDK initialization failed"),
          });
        }
      }
    };
    init();
    return () => controller.abort();
  }, []);

  if (state.phase === "error") return <LoadError message={state.error.message} />;
  if (state.phase === "loading") return <div>Connecting to platform...</div>;

  return (
    <SDKContext.Provider value={{ sdk: state.sdk, user: state.user, isReady: true, error: null }}>
      <PlatformSDKRuntime>{children}</PlatformSDKRuntime>
    </SDKContext.Provider>
  );
}
```

`PlatformSDKRuntime` calls `useHostRouter()` inside the provider so back-button sync is always active. The provider is **the only place the SDK is wrapped**; children consume it via `usePlatformSDK()`.

### `useHostRouter` - bidirectional navigation sync

Keep the host's back-button policy in sync and handle `navigation.back.requested`:

```typescript
// src/hooks/useHostRouter.ts
export const NAVIGATION_EVENTS = {
  BACK_REQUESTED: "navigation.back.requested",
  ROUTE_CHANGED: "navigation.route.changed",
} as const;

export function useHostRouter() {
  const { sdk, isReady } = usePlatformSDK();
  const navigate = useNavigate();
  const location = useLocation();
  const navigationType = useNavigationType();

  // With HashRouter the pathname IS the hash path (e.g. "#/chat" -> "/chat").
  const isRoot = location.pathname === "/" || location.pathname === "";
  const canGoBackRef = useRef(!isRoot);
  canGoBackRef.current = !isRoot;

  const lastKeyRef = useRef<string | null>(null);
  const previousPathRef = useRef(location.pathname);

  useEffect(() => {
    if (lastKeyRef.current === location.key) return;

    const isFirstNavigation = lastKeyRef.current === null;
    lastKeyRef.current = location.key;

    if (isFirstNavigation) {
      previousPathRef.current = location.pathname;
      if (!isRoot && sdk) {
        sdk.emit(NAVIGATION_EVENTS.ROUTE_CHANGED, {
          previous: previousPathRef.current,
          current: location.pathname,
          canGoBack: true,
        });
        void sdk.navigation.router?.push?.(true);
      }
      return;
    }

    if (!sdk) return;

    const canGoBack = !isRoot;
    sdk.emit(NAVIGATION_EVENTS.ROUTE_CHANGED, {
      previous: previousPathRef.current,
      current: location.pathname,
      canGoBack,
    });
    previousPathRef.current = location.pathname;
    if (!isRoot && navigationType === "PUSH") {
      void sdk.navigation.router?.push?.(true);
    }
  }, [location.key, location.pathname, navigationType, sdk, isRoot]);

  const goBack = useCallback(async () => {
    const consumed = canGoBackRef.current;
    await sdk?.navigation.router?.back?.(consumed);
    if (consumed) navigate(-1);
  }, [navigate, sdk]);

  // Sync host when SDK becomes ready; the SDK-ready effect re-emits ROUTE_CHANGED.
  useEffect(() => {
    if (!sdk || !isReady) return;
    if (!isRoot) {
      sdk.emit(NAVIGATION_EVENTS.ROUTE_CHANGED, {
        previous: previousPathRef.current,
        current: location.pathname,
        canGoBack: true,
      });
      void sdk.navigation.router?.push?.(true);
    }
  }, [sdk, isReady, location.pathname, isRoot]);

  useEffect(() => {
    if (!sdk || !isReady) return;
    return sdk.on(NAVIGATION_EVENTS.BACK_REQUESTED, () => { void goBack(); });
  }, [sdk, isReady, goBack]);
}
```

Key difference vs. the simplified version: instead of maintaining a `depthRef` counter, it derives `canGoBack` from whether the current route is the root. Router calls are guarded with `?.` because some mobile hosts don't implement the router bridge — absence of a bridge must not crash rendering.

## 5. Vite Build

```typescript
// src/utils/createManifest.ts
import tailwindcss from "@tailwindcss/vite";
import react from "@vitejs/plugin-react";
import { defineConfig } from "vite";
import { createManifest } from "./src/utils/createManifest.ts";

export default defineConfig({
  plugins: [
    react({ jsxRuntime: "automatic" }),
    tailwindcss(),
    createManifest(),
  ],
  define: { "process.env.NODE_ENV": JSON.stringify("production") },
  build: {
    target: "chrome89",
    manifest: "manifest.json",
    outDir: "dist",
    cssCodeSplit: true,
    lib: {
      entry: "./src/main.tsx",
      name: "RevenueMiniApp",
      formats: ["es"],
      fileName: () => "[name][hash].js",
    },
    rollupOptions: {
      output: {
        codeSplitting: false,
        format: "es",
        chunkFileNames: "assets/[name]-[hash].js",
        assetFileNames: "assets/[name]-[hash][extname]",
      },
    },
  },
  server: {
    host: "0.0.0.0",
    cors: true,
    headers: { "Access-Control-Allow-Origin": "*" },
  },
});
```

The `createManifest` plugin (src/utils/createManifest.ts:8) reads Vite's `.vite/manifest.json`, finds the entry, collects all `dist` files, and writes a host-discovery manifest to `dist/manifest.json`.

### Manifest Schema

The emitted `manifest.json` is the contract the host uses to load your app:

```json
{
  "schemaVersion": "1",
  "framework": "react",
  "application": {
    "id": "remote-app",
    "name": "remote-app",
    "version": "0.0.0",
    "type": "mini-app"
  },
  "platform": {
    "runtime": "^1.0.0",
    "sdk": "^1.0.9"
  },
  "bundle": {
    "entry": "RevenueMiniApp.<hash>.js",
    "styles": ["RevenueMiniApp.<hash>.css"],
    "files": ["RevenueMiniApp.<hash>.js", "RevenueMiniApp.<hash>.css"]
  }
}
```

- `platform.runtime` / `platform.sdk` are semver constraints the host matches against its own runtime and SDK versions. They are **metadata for the host loader**, not a runtime dependency your bundle imports. Your bundle ships zero SDK code — the host injects `window.__GSA_SDK__` at runtime.
- `bundle.entry` points to the ES module the host loads (`lib.name` + hash).
- `bundle.files` lists every asset in `dist` so the host knows the full payload.

> **Is the runtime needed to initialize the app?** No. The Vite bundle needs no runtime/bridge import. Only the **SDK global injected by the host** is required; the `platform` fields are host-side version constraints, not a bundle dependency.
